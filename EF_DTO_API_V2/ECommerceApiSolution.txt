Product listing should be cache and also use HTTPResponseCache. 
Please also include NUnit test cases ans console program to consume these APIs.

```csharp
public interface IProductService
{
    Task<List<ProductDTO>> GetAllAsync();
    Task<ProductDTO> GetByIdAsync(int id);
    Task<ProductDTO> CreateAsync(ProductDTO product);
    Task UpdateAsync(int id, ProductDTO product);
    Task DeleteAsync(int id);
}

// ProductService.cs
public class ProductService : IProductService
{
    private readonly AppDbContext _context;
    private readonly IMemoryCache _cache;
    private readonly string _cacheKey = "ProductList";

    public ProductService(AppDbContext context, IMemoryCache cache)
    {
        _context = context;
        _cache = cache;
    }

    public async Task<List<ProductDTO>> GetAllAsync()
    {
        if (!_cache.TryGetValue(_cacheKey, out List<ProductDTO> products))
        {
            products = await _context.Products
                .FromSqlRaw("EXEC sp_GetAllProducts")
                .Select(p => new ProductDTO
                {
                    Id = p.Id,
                    Name = p.Name,
                    Price = p.Price,
                    CompanyId = p.CompanyId
                })
                .ToListAsync();

            var cacheOptions = new MemoryCacheEntryOptions()
                .SetSlidingExpiration(TimeSpan.FromMinutes(30));
            _cache.Set(_cacheKey, products, cacheOptions);
        }
        return products;
    }

    public async Task<ProductDTO> GetByIdAsync(int id)
    {
        var product = await _context.Products
            .FromSqlRaw("EXEC sp_GetProductById @Id", new SqlParameter("@Id", id))
            .Select(p => new ProductDTO
            {
                Id = p.Id,
                Name = p.Name,
                Price = p.Price,
                CompanyId = p.CompanyId
            })
            .FirstOrDefaultAsync();

        return product ?? throw new KeyNotFoundException("Product not found");
    }

    public async Task<ProductDTO> CreateAsync(ProductDTO product)
    {
        var parameters = new[]
        {
            new SqlParameter("@Name", product.Name),
            new SqlParameter("@Price", product.Price),
            new SqlParameter("@CompanyId", product.CompanyId)
        };

        await _context.Database.ExecuteSqlRawAsync("EXEC sp_CreateProduct @Name, @Price, @CompanyId", parameters);
        _cache.Remove(_cacheKey);
        return product;
    }

    public async Task UpdateAsync(int id, ProductDTO product)
    {
        var parameters = new[]
        {
            new SqlParameter("@Id", id),
            new SqlParameter("@Name", product.Name),
            new SqlParameter("@Price", product.Price),
            new SqlParameter("@CompanyId", product.CompanyId)
        };

        await _context.Database.ExecuteSqlRawAsync("EXEC sp_UpdateProduct @Id, @Name, @Price, @CompanyId", parameters);
        _cache.Remove(_cacheKey);
    }

    public async Task DeleteAsync(int id)
    {
        var parameter = new SqlParameter("@Id", id);
        await _context.Database.ExecuteSqlRawAsync("EXEC sp_DeleteProduct @Id", parameter);
        _cache.Remove(_cacheKey);
    }
}

// OrderDetailService.cs
public class OrderDetailService : IOrderDetailService
{
    private readonly AppDbContext _context;

    public OrderDetailService(AppDbContext context)
    {
        _context = context;
    }

    public async Task<List<OrderDetailDTO>> GetAllAsync()
    {
        return await _context.OrderDetails
            .FromSqlRaw("EXEC sp_GetAllOrderDetails")
            .Select(od => new OrderDetailDTO
            {
                Id = od.Id,
                OrderId = od.OrderId,
                ProductId = od.ProductId,
                Quantity = od.Quantity,
                TotalCost = od.TotalCost
            })
            .ToListAsync();
    }

    public async Task<OrderDetailDTO> GetByIdAsync(int id)
    {
        var orderDetail = await _context.OrderDetails
            .FromSqlRaw("EXEC sp_GetOrderDetailById @Id", new SqlParameter("@Id", id))
            .Select(od => new OrderDetailDTO
            {
                Id = od.Id,
                OrderId = od.OrderId,
                ProductId = od.ProductId,
                Quantity = od.Quantity,
                TotalCost = od.TotalCost
            })
            .FirstOrDefaultAsync();

        return orderDetail ?? throw new KeyNotFoundException("OrderDetail not found");
    }

    public async Task<OrderDetailDTO> CreateAsync(OrderDetailDTO orderDetail)
    {
        // Calculate TotalCost (Quantity * Product.Price)
        var product = await _context.Products.FindAsync(orderDetail.ProductId);
        if (product == null) throw new KeyNotFoundException("Product not found");
        orderDetail.TotalCost = orderDetail.Quantity * product.Price;

        var parameters = new[]
        {
            new SqlParameter("@OrderId", orderDetail.OrderId),
            new SqlParameter("@ProductId", orderDetail.ProductId),
            new SqlParameter("@Quantity", orderDetail.Quantity),
            new SqlParameter("@TotalCost", orderDetail.TotalCost)
        };

        await _context.Database.ExecuteSqlRawAsync("EXEC sp_CreateOrderDetail @OrderId, @ProductId, @Quantity, @TotalCost", parameters);

        // Update Order TotalCost
        await UpdateOrderTotalCost(orderDetail.OrderId);

        return orderDetail;
    }

    public async Task UpdateAsync(int id, OrderDetailDTO orderDetail)
    {
        // Calculate TotalCost (Quantity * Product.Price)
        var product = await _context.Products.FindAsync(orderDetail.ProductId);
        if (product == null) throw new KeyNotFoundException("Product not found");
        orderDetail.TotalCost = orderDetail.Quantity * product.Price;

        var parameters = new[]
        {
            new SqlParameter("@Id", id),
            new SqlParameter("@OrderId", orderDetail.OrderId),
            new SqlParameter("@ProductId", orderDetail.ProductId),
            new SqlParameter("@Quantity", orderDetail.Quantity),
            new SqlParameter("@TotalCost", orderDetail.TotalCost)
        };

        await _context.Database.ExecuteSqlRawAsync("EXEC sp_UpdateOrderDetail @Id, @OrderId, @ProductId, @Quantity, @TotalCost", parameters);

        // Update Order TotalCost
        await UpdateOrderTotalCost(orderDetail.OrderId);
    }

    public async Task DeleteAsync(int id)
    {
        var orderDetail = await _context.OrderDetails.FindAsync(id);
        if (orderDetail == null) throw new KeyNotFoundException("OrderDetail not found");

        var parameter = new SqlParameter("@Id", id);
        await _context.Database.ExecuteSqlRawAsync("EXEC sp_DeleteOrderDetail @Id", parameter);

        // Update Order TotalCost
        await UpdateOrderTotalCost(orderDetail.OrderId);
    }

    private async Task UpdateOrderTotalCost(int orderId)
    {
        var totalCost = await _context.OrderDetails
            .Where(od => od.OrderId == orderId)
            .SumAsync(od => od.TotalCost);

        var parameters = new[]
        {
            new SqlParameter("@Id", orderId),
            new SqlParameter("@TotalCost", totalCost)
        };

        await _context.Database.ExecuteSqlRawAsync("EXEC sp_UpdateOrderTotalCost @Id, @TotalCost", parameters);
    }
}
```

```csharp
// TokenController.cs
[Route("api/[controller]")]
[ApiController]
public class TokenController : ControllerBase
{
    private readonly IConfiguration _configuration;

    public TokenController(IConfiguration configuration)
    {
        _configuration = configuration;
    }

    [HttpPost]
    public IActionResult GetToken([FromBody] TokenRequest request)
    {
        // Simplified token generation - in production, use a proper Identity Provider
        if (request.GrantType != "client_credentials" ||
            request.ClientId != _configuration["Authentication:ClientId"] ||
            request.ClientSecret != _configuration["Authentication:ClientSecret"])
        {
            return Unauthorized();
        }

        var token = GenerateJwtToken();
        return Ok(new TokenResponse
        {
            AccessToken = token,
            ExpiresIn = 120, // 2 minutes
            TokenType = "Bearer"
        });
    }

    private string GenerateJwtToken()
    {
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_configuration["Authentication:JwtKey"]));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: _configuration["Authentication:Issuer"],
            audience: _configuration["Authentication:Audience"],
            expires: DateTime.Now.AddSeconds(120), // 2 minutes
            signingCredentials: creds
        );

        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

### Program.cs (.NET 8)

```csharp
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.EntityFrameworkCore;
using Microsoft.IdentityModel.Tokens;
using System.Text;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));

builder.Services.AddMemoryCache();
builder.Services.AddResponseCaching();

builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidateAudience = true,
            ValidateLifetime = true,
            ValidateIssuerSigningKey = true,
            ValidIssuer = builder.Configuration["Authentication:Issuer"],
            ValidAudience = builder.Configuration["Authentication:Audience"],
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(builder.Configuration["Authentication:JwtKey"]))
        };
    });

builder.Services.AddSwaggerGen();
builder.Services.AddScoped<IProductService, ProductService>();
builder.Services.AddScoped<IOrderService, OrderService>();
builder.Services.AddScoped<IOrderDetailService, OrderDetailService>();

var app = builder.Build();

app.UseResponseCaching();
app.UseRouting();
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();
app.UseSwagger();
app.UseSwaggerUI();

app.Run();
```

### StoredProcedures.sql

```sql
CREATE PROCEDURE sp_GetAllProducts
AS
BEGIN
    SELECT * FROM Products
END

CREATE PROCEDURE sp_GetProductById
    @Id INT
AS
BEGIN
    SELECT * FROM Products WHERE Id = @Id
END

CREATE PROCEDURE sp_CreateProduct
    @Name NVARCHAR(100),
    @Price DECIMAL(18,2),
    @CompanyId INT
AS
BEGIN
    INSERT INTO Products (Name, Price, CompanyId)
    VALUES (@Name, @Price, @CompanyId)
END

CREATE PROCEDURE sp_UpdateProduct
    @Id INT,
    @Name NVARCHAR(100),
    @Price DECIMAL(18,2),
    @CompanyId INT
AS
BEGIN
    UPDATE Products
    SET Name = @Name, Price = @Price, CompanyId = @CompanyId
    WHERE Id = @Id
END

CREATE PROCEDURE sp_DeleteProduct
    @Id INT
AS
BEGIN
    DELETE FROM Products WHERE Id = @Id
END

CREATE PROCEDURE sp_GetAllOrders
AS
BEGIN
    SELECT * FROM Orders
END

CREATE PROCEDURE sp_GetOrderById
    @Id INT
AS
BEGIN
    SELECT * FROM Orders WHERE Id = @Id
END

CREATE PROCEDURE sp_CreateOrder
    @UserId INT,
    @OrderDate DATETIME,
    @TotalCost DECIMAL(18,2)
AS
```

```csharp
public class Program
{
    private static readonly HttpClient _client = new HttpClient();

    public static async Task Main()
    {
        var token = await GetTokenAsync();
        _client.DefaultRequestHeaders.Authorization = 
            new AuthenticationHeaderValue("Bearer", token);

        // Create a product
        var product = new ProductDTO { Name = "Laptop", Price = 999.99m, CompanyId = 1 };
        var productResponse = await _client.PostAsJsonAsync("https://localhost:5001/api/product", product);
        var createdProduct = await productResponse.Content.ReadFromJsonAsync<ProductDTO>();
        Console.WriteLine($"Created Product: {createdProduct.Name}");

        // Create an order
        var order = new OrderDTO { UserId = 1, OrderDate = DateTime.Now, TotalCost = 0 };
        var orderResponse = await _client.PostAsJsonAsync("https://localhost:5001/api/order", order);
        var createdOrder = await orderResponse.Content.ReadFromJsonAsync<OrderDTO>();
        Console.WriteLine($"Created Order: {createdOrder.Id}");

        // Create an order detail
        var orderDetail = new OrderDetailDTO { OrderId = createdOrder.Id, ProductId = createdProduct.Id, Quantity = 2 };
        var orderDetailResponse = await _client.PostAsJsonAsync("https://localhost:5001/api/orderdetail", orderDetail);
        var createdOrderDetail = await orderDetailResponse.Content.ReadFromJsonAsync<OrderDetailDTO>();
        Console.WriteLine($"Created OrderDetail: {createdOrderDetail.Id} with TotalCost: ${createdOrderDetail.TotalCost}");

        // Get products
        var productsResponse = await _client.GetAsync("https://localhost:5001/api/product");
        var products = await productsResponse.Content.ReadFromJsonAsync<List<ProductDTO>>();
        Console.WriteLine("Products:");
        products.ForEach(p => Console.WriteLine($"{p.Name}: ${p.Price}"));

        // Get order details
        var orderDetailsResponse = await _client.GetAsync("https://localhost:5001/api/orderdetail");
        var orderDetails = await orderDetailsResponse.Content.ReadFromJsonAsync<List<OrderDetailDTO>>();
        Console.WriteLine("Order Details:");
        orderDetails.ForEach(od => Console.WriteLine($"Order {od.OrderId}, Product {od.ProductId}, Total: ${od.TotalCost}"));
    }

    private static async Task<string> GetTokenAsync()
    {
        var tokenRequest = new TokenRequest
        {
            GrantType = "client_credentials",
            ClientId = "your-client-id",
            ClientSecret = "your-client-secret",
            Scope = "your-api"
        };

        var response = await _client.PostAsJsonAsync("https://localhost:5001/api/token", tokenRequest);
        var tokenResponse = await response.Content.ReadFromJsonAsync<TokenResponse>();
        return tokenResponse.AccessToken;
    }
}
```
